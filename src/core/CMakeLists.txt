add_library(vv_core STATIC
  common.cpp telemetry.cpp config.cpp http_client.cpp devices.cpp sources.cpp wasapi_source.cpp
  video_wgc.cpp video_dxgi.cpp encoder.cpp recorder.cpp quality.cpp share_copy.cpp control.cpp app_controller.cpp
  redact.cpp secrets.cpp crash_handler.cpp)

target_include_directories(vv_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_SOURCE_DIR}/third_party)
target_link_libraries(vv_core PUBLIC vv_ffmpeg
  d3d11 dxgi dwmapi ole32 oleaut32 mmdevapi winmm shell32 user32 windowsapp dbghelp crypt32 winhttp)

# The C++/WinRT bundled with the VS2019 SDK predates standard coroutines; use a current projection
# (generated by scripts/setup-deps.ps1) and the standard <coroutine> that /std:c++latest enables.
set(VV_CPPWINRT_INCLUDE "" CACHE PATH "Generated C++/WinRT projection headers")
if(VV_CPPWINRT_INCLUDE)
  set_source_files_properties(video_wgc.cpp PROPERTIES INCLUDE_DIRECTORIES ${VV_CPPWINRT_INCLUDE})
endif()
set_source_files_properties(video_wgc.cpp PROPERTIES COMPILE_OPTIONS "/std:c++latest")
