find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)

set(VV_UI_SOURCES main.cpp panel.cpp tray.cpp theme.cpp)
if(EXISTS ${CMAKE_SOURCE_DIR}/assets/icon.ico)  # generated by scripts/make-icon.py
  list(APPEND VV_UI_SOURCES app.rc)
endif()
qt_add_executable(voloire-video WIN32 ${VV_UI_SOURCES})
target_link_libraries(voloire-video PRIVATE vv_core Qt6::Widgets)

# Fonts and Qt runtime next to the binary so the build tree runs as is.
add_custom_command(TARGET voloire-video POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets/fonts $<TARGET_FILE_DIR:voloire-video>/fonts)

get_target_property(_qmake Qt6::qmake IMPORTED_LOCATION)
get_filename_component(_qtbin ${_qmake} DIRECTORY)
find_program(VV_WINDEPLOYQT windeployqt HINTS ${_qtbin} NO_DEFAULT_PATH)
if(VV_WINDEPLOYQT)
  add_custom_command(TARGET voloire-video POST_BUILD
    COMMAND ${VV_WINDEPLOYQT} --no-translations --no-system-d3d-compiler --no-opengl-sw --no-quick-import
            --dir $<TARGET_FILE_DIR:voloire-video> $<TARGET_FILE:voloire-video>)
endif()
